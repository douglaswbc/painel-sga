-- 1. CONFIGURAÇÃO E CRIPTOGRAFIA

CREATE TABLE IF NOT EXISTS public.app_config (
  key text PRIMARY KEY,
  value text NOT NULL
);

REVOKE ALL ON TABLE public.app_config FROM PUBLIC, anon, authenticated;
GRANT SELECT, INSERT, UPDATE, DELETE ON TABLE public.app_config TO service_role;

CREATE OR REPLACE FUNCTION public.encrypt_val(input_text text)
RETURNS bytea
LANGUAGE plpgsql
VOLATILE
SECURITY DEFINER
SET search_path = ''
AS $$
DECLARE
  enc_key text;
  encrypted_value bytea;
BEGIN
  IF input_text IS NULL THEN RETURN NULL; END IF;
  SELECT value INTO enc_key FROM public.app_config WHERE key = 'encryption_key';
  IF enc_key IS NULL THEN RAISE EXCEPTION 'Chave de criptografia não configurada.'; END IF;
  SELECT extensions.pgp_sym_encrypt(input_text, enc_key) INTO encrypted_value;
  RETURN encrypted_value;
EXCEPTION WHEN undefined_function THEN
  SELECT pgp_sym_encrypt(input_text, enc_key) INTO encrypted_value;
  RETURN encrypted_value;
END;
$$;

CREATE OR REPLACE FUNCTION public.decrypt_val(input_bytea bytea)
RETURNS text
LANGUAGE plpgsql
VOLATILE
SECURITY DEFINER
SET search_path = ''
AS $$
DECLARE
  enc_key text;
  decrypted_value text;
BEGIN
  IF input_bytea IS NULL THEN RETURN NULL; END IF;
  SELECT value INTO enc_key FROM public.app_config WHERE key = 'encryption_key';
  IF enc_key IS NULL THEN RAISE EXCEPTION 'Chave de criptografia não configurada.'; END IF;
  SELECT extensions.pgp_sym_decrypt(input_bytea, enc_key) INTO decrypted_value;
  RETURN decrypted_value;
EXCEPTION WHEN undefined_function THEN
  SELECT pgp_sym_decrypt(input_bytea, enc_key) INTO decrypted_value;
  RETURN decrypted_value;
END;
$$;

REVOKE ALL ON FUNCTION public.encrypt_val(text) FROM PUBLIC, anon, authenticated;
GRANT EXECUTE ON FUNCTION public.encrypt_val(text) TO service_role;

REVOKE ALL ON FUNCTION public.decrypt_val(bytea) FROM PUBLIC, anon;
GRANT EXECUTE ON FUNCTION public.decrypt_val(bytea) TO authenticated, service_role;

-- Tabela de Usuários (detalhes do usuário autenticado / locador / admin)
CREATE TABLE IF NOT EXISTS public.usuarios (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  nome_completo text,
  whatsapp text UNIQUE,
  email text NOT NULL UNIQUE,
  role text NOT NULL,
  empresa_nome text,
  ativo boolean NOT NULL DEFAULT true,
  created_at timestamptz NOT NULL DEFAULT now(),
  mp_notification_url text,
  reminder_stagger_minutes integer DEFAULT 2
    CHECK (reminder_stagger_minutes >= 0 AND reminder_stagger_minutes <= 60),
  reminder_dispatch_delay_seconds integer DEFAULT 5
    CHECK (reminder_dispatch_delay_seconds >= 0 AND reminder_dispatch_delay_seconds <= 30)
);

-- Tabela de Locatários
CREATE TABLE IF NOT EXISTS public.locatarios (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  id_usuario uuid NOT NULL REFERENCES public.usuarios(id) ON DELETE CASCADE,
  nome_completo text NOT NULL,
  whatsapp text NOT NULL,
  email text,
  documento text,
  ativo boolean NOT NULL DEFAULT true,
  created_at timestamptz NOT NULL DEFAULT now(),
  cpf text,
  cep text,
  rua text,
  numero text,
  bairro text,
  cidade text,
  estado character(2),
  portal_token uuid DEFAULT gen_random_uuid(),
  documento_enc bytea,
  cpf_enc bytea
);

CREATE UNIQUE INDEX IF NOT EXISTS locatarios_id_usuario_uidx
  ON public.locatarios (id, id_usuario);

-- View segura de Locatários (com resolução transparente de campos criptografados)
CREATE OR REPLACE VIEW public.locatarios_safe
WITH (security_invoker = true)
AS
SELECT
  id,
  id_usuario,
  nome_completo,
  whatsapp,
  email,
  COALESCE(public.decrypt_val(documento_enc), documento) AS documento,
  COALESCE(public.decrypt_val(cpf_enc), cpf) AS cpf,
  ativo,
  created_at,
  cep,
  rua,
  numero,
  bairro,
  cidade,
  estado,
  portal_token,
  documento_enc,
  cpf_enc
FROM public.locatarios;

-- View Unificada de Atividade e Auditoria de WhatsApp
CREATE OR REPLACE VIEW public.zernio_whatsapp_activity_log
WITH (security_invoker = true)
AS
SELECT d.id_usuario,
       'charge'::text AS type,
       d.id AS source_id,
       d.status,
       d.template_name,
       d.template_language,
       t.category,
       l.nome_completo AS recipient_name,
       l.whatsapp AS recipient_phone,
       COALESCE(d.sent_at, d.updated_at, d.created_at) AS occurred_at,
       d.created_at,
       d.message_id,
       COALESCE(d.safe_error, d.error_code) AS error,
       d.meta_status
FROM public.zernio_whatsapp_deliveries d
LEFT JOIN public.zernio_whatsapp_templates t
  ON t.id = d.template_id AND t.id_usuario = d.id_usuario
LEFT JOIN public.cobrancas c
  ON c.id = d.id_cobranca AND c.id_usuario = d.id_usuario
LEFT JOIN public.locatarios l
  ON l.id = c.id_locatario AND l.id_usuario = d.id_usuario
UNION ALL
SELECT d.id_usuario,
       'scheduled'::text,
       d.id,
       d.status,
       t.name,
       t.language,
       t.category,
       l.nome_completo,
       l.whatsapp,
       COALESCE(d.sent_at, d.attempted_at, d.created_at),
       d.created_at,
       d.message_id,
       COALESCE(d.safe_error, d.error_code),
       d.meta_status
FROM public.zernio_whatsapp_scheduled_deliveries d
JOIN public.agendamentos_mensagens a
  ON a.id = d.id_agendamento AND a.id_usuario = d.id_usuario
LEFT JOIN public.zernio_whatsapp_templates t
  ON t.id = a.zernio_template_id AND t.id_usuario = d.id_usuario
LEFT JOIN public.locatarios l
  ON l.id = a.id_locatario AND l.id_usuario = d.id_usuario
UNION ALL
SELECT d.id_usuario,
       'payment'::text,
       d.id,
       d.status,
       t.name,
       t.language,
       t.category,
       COALESCE(l.nome_completo, cand.nome),
       COALESCE(l.whatsapp, cand.telefone),
       COALESCE(d.updated_at, d.created_at),
       d.created_at,
       d.message_id,
       d.safe_error,
       d.meta_status
FROM public.zernio_payment_notifications d
LEFT JOIN public.cobrancas c
  ON c.id = d.id_cobranca AND c.id_usuario = d.id_usuario
LEFT JOIN public.locatarios l
  ON l.id = c.id_locatario AND l.id_usuario = d.id_usuario
LEFT JOIN public.candidatos cand
  ON cand.id = d.id_candidato AND cand.id_usuario = d.id_usuario
LEFT JOIN public.zernio_whatsapp_template_usages u
  ON u.id_usuario = d.id_usuario AND u.purpose = 'payment_confirmation' AND u.is_default
LEFT JOIN public.zernio_whatsapp_templates t
  ON t.id = u.template_id AND t.id_usuario = d.id_usuario
UNION ALL
SELECT d.id_usuario,
       'manual'::text,
       d.id,
       d.status,
       d.template_name,
       d.template_language,
       t.category,
       l.nome_completo,
       l.whatsapp,
       COALESCE(d.sent_at, d.updated_at, d.created_at),
       d.created_at,
       d.message_id,
       COALESCE(d.safe_error, d.error_code),
       d.meta_status
FROM public.zernio_whatsapp_manual_deliveries d
LEFT JOIN public.zernio_whatsapp_templates t
  ON t.id = d.template_id AND t.id_usuario = d.id_usuario
LEFT JOIN public.locatarios l
  ON l.id = d.id_locatario AND l.id_usuario = d.id_usuario;

-- 6.2 Verificação de Perfil Administrador
CREATE OR REPLACE FUNCTION public.check_is_admin()
RETURNS boolean
LANGUAGE sql
STABLE
SECURITY DEFINER
SET search_path = ''
AS $$
  SELECT EXISTS (
    SELECT 1 FROM public.usuarios u
    WHERE u.id = (SELECT auth.uid()) AND u.role = 'admin'
  );
$$;

REVOKE ALL ON FUNCTION public.check_is_admin() FROM PUBLIC, anon;
GRANT EXECUTE ON FUNCTION public.check_is_admin() TO authenticated, service_role;

-- 6.3 Atualização administrativa de usuários
CREATE OR REPLACE FUNCTION public.admin_update_usuario(
  p_user_id uuid,
  p_nome_completo text,
  p_whatsapp text,
  p_role text,
  p_ativo boolean
)
RETURNS void
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $$
BEGIN
  IF NOT public.check_is_admin() THEN
    RAISE EXCEPTION 'Acesso restrito a administradores.' USING ERRCODE = '42501';
  END IF;
  IF p_role NOT IN ('admin', 'locador', 'locatario') THEN
    RAISE EXCEPTION 'Papel de usuário inválido.' USING ERRCODE = '22023';
  END IF;
  UPDATE public.usuarios
  SET nome_completo = p_nome_completo,
      whatsapp = p_whatsapp,
      role = p_role,
      ativo = p_ativo
  WHERE id = p_user_id;
  IF NOT FOUND THEN
    RAISE EXCEPTION 'Usuário não encontrado.' USING ERRCODE = 'P0002';
  END IF;
END;
$$;

CREATE OR REPLACE FUNCTION public.admin_set_usuario_status(p_user_id uuid, p_ativo boolean)
RETURNS void
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $$
BEGIN
  IF NOT public.check_is_admin() THEN
    RAISE EXCEPTION 'Acesso restrito a administradores.' USING ERRCODE = '42501';
  END IF;
  UPDATE public.usuarios SET ativo = p_ativo WHERE id = p_user_id;
  IF NOT FOUND THEN
    RAISE EXCEPTION 'Usuário não encontrado.' USING ERRCODE = 'P0002';
  END IF;
END;
$$;

REVOKE ALL ON FUNCTION public.admin_update_usuario(uuid, text, text, text, boolean) FROM PUBLIC, anon;
REVOKE ALL ON FUNCTION public.admin_set_usuario_status(uuid, boolean) FROM PUBLIC, anon;
GRANT EXECUTE ON FUNCTION public.admin_update_usuario(uuid, text, text, text, boolean) TO authenticated, service_role;
GRANT EXECUTE ON FUNCTION public.admin_set_usuario_status(uuid, boolean) TO authenticated, service_role;

-- 6.4 Criptografia automática de documentos de locatários
CREATE OR REPLACE FUNCTION public.encrypt_locatario_fields()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path TO ''
AS $$
BEGIN
  IF NEW.documento IS NOT NULL AND NEW.documento != '' THEN
    NEW.documento_enc = public.encrypt_val(NEW.documento);
  END IF;
  IF NEW.cpf IS NOT NULL AND NEW.cpf != '' THEN

-- 6.5 Sincronização e Validação do Locatário na Cobrança
CREATE OR REPLACE FUNCTION public.set_cobranca_locatario()
RETURNS trigger
LANGUAGE plpgsql
SET search_path = public, pg_temp
AS $$
DECLARE
  contract_tenant_id uuid;
BEGIN
  IF NEW.id_contrato IS NULL THEN
    RETURN NEW;
  END IF;

  SELECT k.id_locatario INTO contract_tenant_id
  FROM public.contratos k
  WHERE k.id = NEW.id_contrato AND k.id_usuario = NEW.id_usuario;

  IF contract_tenant_id IS NULL THEN
    RAISE EXCEPTION 'O contrato da cobranca nao pertence ao locador informado.';
  END IF;
  IF NEW.id_locatario IS NOT NULL AND NEW.id_locatario <> contract_tenant_id THEN
    RAISE EXCEPTION 'O locatario da cobranca nao corresponde ao contrato.';
  END IF;

  NEW.id_locatario := contract_tenant_id;
  RETURN NEW;
END;
$$;

REVOKE ALL ON FUNCTION public.set_cobranca_locatario() FROM PUBLIC, anon, authenticated;
DROP TRIGGER IF EXISTS trg_set_cobranca_locatario ON public.cobrancas;
CREATE TRIGGER trg_set_cobranca_locatario
  BEFORE INSERT OR UPDATE OF id_contrato, id_usuario, id_locatario ON public.cobrancas
  FOR EACH ROW EXECUTE FUNCTION public.set_cobranca_locatario();

-- 6.6 Resumo Financeiro e Indicadores do Dashboard
CREATE OR REPLACE FUNCTION public.get_dashboard_summary()
RETURNS TABLE (
  saldo_liquido_mes numeric,
  receita_a_receber numeric,
  total_atrasado numeric,
  contratos_ativos bigint,
  adimplencia_mes numeric,
  cobrancas_vencidas_mes bigint,
  cobrancas_pagas_mes bigint
)
LANGUAGE sql
STABLE
SECURITY INVOKER
SET search_path = public
AS $$
  WITH periodo AS (
    SELECT
      date_trunc('month', now() AT TIME ZONE 'America/Sao_Paulo')::date AS inicio_mes,
      (now() AT TIME ZONE 'America/Sao_Paulo')::date AS hoje,
      date_trunc('month', now() AT TIME ZONE 'America/Sao_Paulo') AT TIME ZONE 'America/Sao_Paulo' AS inicio_mes_utc,
      (date_trunc('month', now() AT TIME ZONE 'America/Sao_Paulo') + interval '1 month') AT TIME ZONE 'America/Sao_Paulo' AS inicio_proximo_mes_utc
  ),
  cobrancas_usuario AS (
    SELECT c.*
    FROM public.cobrancas AS c
    WHERE c.id_usuario = (SELECT auth.uid())
  ),
  financeiro AS (
    SELECT
      COALESCE(SUM(c.valor) FILTER (
        WHERE c.tipo = 'receita'
          AND c.status = 'pago'
          AND COALESCE(c.data_pagamento, (c.data_vencimento::timestamp + interval '12 hours') AT TIME ZONE 'America/Sao_Paulo') >= p.inicio_mes_utc
          AND COALESCE(c.data_pagamento, (c.data_vencimento::timestamp + interval '12 hours') AT TIME ZONE 'America/Sao_Paulo') < p.inicio_proximo_mes_utc
      ), 0)
      - COALESCE(SUM(c.valor) FILTER (
        WHERE c.tipo = 'despesa'
          AND c.status = 'pago'
          AND COALESCE(c.data_pagamento, (c.data_vencimento::timestamp + interval '12 hours') AT TIME ZONE 'America/Sao_Paulo') >= p.inicio_mes_utc
          AND COALESCE(c.data_pagamento, (c.data_vencimento::timestamp + interval '12 hours') AT TIME ZONE 'America/Sao_Paulo') < p.inicio_proximo_mes_utc
      ), 0) AS saldo_liquido_mes,
      COALESCE(SUM(c.valor) FILTER (
        WHERE c.tipo = 'receita' AND c.status IN ('pendente', 'atrasado')
      ), 0) AS receita_a_receber,
      COALESCE(SUM(c.valor) FILTER (
        WHERE c.tipo = 'receita' AND c.status = 'atrasado'
      ), 0) AS total_atrasado,
      COUNT(*) FILTER (
        WHERE c.tipo = 'receita'
          AND c.status <> 'cancelado'
          AND c.data_vencimento >= p.inicio_mes
          AND c.data_vencimento <= p.hoje
      ) AS cobrancas_vencidas_mes,
      COUNT(*) FILTER (
        WHERE c.tipo = 'receita'
          AND c.status = 'pago'
          AND c.data_vencimento >= p.inicio_mes
          AND c.data_vencimento <= p.hoje
      ) AS cobrancas_pagas_mes
    FROM cobrancas_usuario AS c
    CROSS JOIN periodo AS p
  ),
  contratos_usuario AS (
    SELECT COUNT(*) AS contratos_ativos
    FROM public.contratos AS c
    WHERE c.id_usuario = (SELECT auth.uid()) AND c.status = 'ativo'
  )
  SELECT
    financeiro.saldo_liquido_mes,
    financeiro.receita_a_receber,
    financeiro.total_atrasado,
    contratos_usuario.contratos_ativos,
    CASE
      WHEN financeiro.cobrancas_vencidas_mes = 0 THEN NULL
      ELSE round(financeiro.cobrancas_pagas_mes::numeric * 100 / financeiro.cobrancas_vencidas_mes, 1)
    END AS adimplencia_mes,
    financeiro.cobrancas_vencidas_mes,
    financeiro.cobrancas_pagas_mes
  FROM financeiro
  CROSS JOIN contratos_usuario;
$$;

REVOKE ALL ON FUNCTION public.get_dashboard_summary() FROM PUBLIC;
GRANT EXECUTE ON FUNCTION public.get_dashboard_summary() TO authenticated;

-- 6.7 Confirmação Manual de Reserva de Candidato com Auditoria
CREATE OR REPLACE FUNCTION public.confirmar_reserva_candidato_manual(p_candidato_id uuid, p_motivo text)
RETURNS uuid
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public, auth
AS $$
DECLARE
  v_user_id uuid := auth.uid();
  v_candidate public.candidatos%ROWTYPE;
  v_audit_id uuid;
  v_payment_status text;
BEGIN
  IF v_user_id IS NULL THEN
    RAISE EXCEPTION 'Autenticação obrigatória.' USING ERRCODE = '42501';
  END IF;
  IF p_motivo IS NULL OR char_length(btrim(p_motivo)) < 10 OR char_length(btrim(p_motivo)) > 1000 THEN
    RAISE EXCEPTION 'Informe um motivo entre 10 e 1000 caracteres.' USING ERRCODE = '22023';
  END IF;

  SELECT * INTO v_candidate
  FROM public.candidatos
  WHERE id = p_candidato_id AND id_usuario = v_user_id
  FOR UPDATE;
  IF NOT FOUND THEN
    RAISE EXCEPTION 'Candidato não encontrado.' USING ERRCODE = 'P0002';
  END IF;

  IF lower(btrim(COALESCE(v_candidate.status, ''))) NOT IN ('aguardando reserva', 'pré-aprovado', 'pre-aprovado') THEN
    RAISE EXCEPTION 'O status atual não permite confirmar a reserva.' USING ERRCODE = '22023';
  END IF;
  v_payment_status := lower(btrim(COALESCE(v_candidate.pagamento_status, '')));
  IF v_payment_status IN ('pendente', 'processando', 'autorizado', 'em análise', 'em analise', 'em disputa')
    AND (v_candidate.reserva_expira_em IS NULL OR v_candidate.reserva_expira_em > now()) THEN
    RAISE EXCEPTION 'Há um pagamento PIX ainda ativo; aguarde a reconciliação antes da confirmação manual.' USING ERRCODE = '22023';
  END IF;

  INSERT INTO public.candidato_reserva_auditoria
    (id_candidato, id_usuario, operador_id, acao, origem, status_anterior, status_novo, motivo)
  VALUES
    (v_candidate.id, v_user_id, v_user_id, 'confirmacao_manual', 'operador', v_candidate.status, 'Vaga reservada', btrim(p_motivo))
  RETURNING id INTO v_audit_id;

  UPDATE public.candidatos
  SET status = 'Vaga reservada',
      pagamento_status = 'Confirmado manualmente',
      reserva_confirmada_em = now(),
      updated_at = now()
  WHERE id = v_candidate.id;

  RETURN v_audit_id;
END;
$$;

REVOKE ALL ON FUNCTION public.confirmar_reserva_candidato_manual(uuid, text) FROM PUBLIC, anon;
GRANT EXECUTE ON FUNCTION public.confirmar_reserva_candidato_manual(uuid, text) TO authenticated;

-- 6.8 Auditoria Automática de Transição de Status em Candidatos
CREATE OR REPLACE FUNCTION public.registrar_historico_status_candidato()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public, auth
AS $$
DECLARE
  v_operator_id uuid := auth.uid();
  v_action text;
  v_reason text;
BEGIN
  IF current_setting('app.skip_candidate_reserva_audit', true) = '1' THEN
    RETURN NEW;
  END IF;

  IF OLD.status IS NOT DISTINCT FROM NEW.status
    AND OLD.pagamento_status IS NOT DISTINCT FROM NEW.pagamento_status THEN
    RETURN NEW;
  END IF;

  v_action := CASE
    WHEN OLD.pagamento_status IS DISTINCT FROM NEW.pagamento_status THEN 'atualizacao_pagamento'
    ELSE 'alteracao_status'
  END;
  v_reason := CASE
    WHEN OLD.pagamento_status IS DISTINCT FROM NEW.pagamento_status
      THEN format('Pagamento: %s → %s', COALESCE(OLD.pagamento_status, 'sem status'), COALESCE(NEW.pagamento_status, 'sem status'))
    ELSE format('Status: %s → %s', COALESCE(OLD.status, 'sem status'), COALESCE(NEW.status, 'sem status'))
  END;

  INSERT INTO public.candidato_reserva_auditoria (
    id_candidato, id_usuario, operador_id, acao, origem,
    status_anterior, status_novo,
    pagamento_status_anterior, pagamento_status_novo,
    motivo
  ) VALUES (
    NEW.id, NEW.id_usuario, v_operator_id, v_action,
    CASE WHEN v_operator_id IS NULL THEN 'sistema' ELSE 'operador' END,
    OLD.status, NEW.status,
    OLD.pagamento_status, NEW.pagamento_status,
    v_reason
  );

  RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS candidatos_status_audit ON public.candidatos;
CREATE TRIGGER candidatos_status_audit
  AFTER UPDATE OF status, pagamento_status ON public.candidatos
  FOR EACH ROW
  EXECUTE FUNCTION public.registrar_historico_status_candidato();

-- 6.9 Consumo de Cota de Submissão de Inscrições Públicas
CREATE OR REPLACE FUNCTION public.consumir_cota_inscricao_candidato(p_link_id uuid)
RETURNS boolean
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_day date := (now() AT TIME ZONE 'UTC')::date;
  v_consumed boolean := false;
BEGIN
  UPDATE public.candidate_application_links
  SET requests_day = v_day,
      requests_today = CASE WHEN requests_day IS DISTINCT FROM v_day THEN 1 ELSE requests_today + 1 END,
      updated_at = now()
  WHERE id = p_link_id
    AND ativo
    AND expires_at > now()
    AND (requests_day IS DISTINCT FROM v_day OR requests_today < 50)
  RETURNING true INTO v_consumed;

  RETURN COALESCE(v_consumed, false);
END;
$$;

REVOKE ALL ON FUNCTION public.consumir_cota_inscricao_candidato(uuid) FROM PUBLIC, anon, authenticated;
GRANT EXECUTE ON FUNCTION public.consumir_cota_inscricao_candidato(uuid) TO service_role;
