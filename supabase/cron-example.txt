-- Faturamento automático do SGA
-- Habilite pg_cron, pg_net e Vault no Supabase antes de configurar o job.
-- Guarde valores reais no Vault; não coloque service_role em arquivos versionados.

-- Execute uma vez, trocando os dois valores. Escolha nomes de secrets estáveis.
-- Use the Secret API Key named default in Settings > API Keys.
-- Rotate any key exposed in the failed SQL/error output before using its replacement.
-- In the Supabase Vault UI, store the rotated value as sga_cron_api_key.
-- Do not paste the API key into SQL; the preflight below checks only the Vault name.
select vault.create_secret('https://ykvmzelwpodjmvoujumf.supabase.co', 'sga_project_url')
where not exists (select 1 from vault.secrets where name = 'sga_project_url');

do $$
begin
  if not exists (select 1 from vault.decrypted_secrets where name = 'sga_project_url' and nullif(decrypted_secret, '') is not null) then
    raise exception 'Vault secret sga_project_url is missing or empty.';
  end if;
  if not exists (select 1 from vault.decrypted_secrets where name = 'sga_cron_api_key' and nullif(decrypted_secret, '') is not null) then
    raise exception 'Vault secret sga_cron_api_key is missing or empty.';
  end if;
end;
$$;

-- Recurring sends for Meta approved templates through Zernio.
select cron.unschedule(jobid)
from cron.job
where jobname = 'sga-zernio-message-scheduler';

select cron.schedule(
  'sga-zernio-message-scheduler',
  '*/5 * * * *',
  $$
    select net.http_post(
      url := (select decrypted_secret from vault.decrypted_secrets where name = 'sga_project_url')
             || '/functions/v1/zernio-message-scheduler',
      headers := jsonb_build_object(
        'Content-Type', 'application/json',
        'apikey', (select decrypted_secret from vault.decrypted_secrets where name = 'sga_cron_api_key')
      ),
      body := '{}'::jsonb
    ) as request_id;
  $$
);
