-- Faturamento automático do SGA
-- Habilite pg_cron, pg_net e Vault no Supabase antes de configurar o job.
-- Guarde valores reais no Vault; não coloque service_role em arquivos versionados.

-- Execute uma vez, trocando os dois valores. Escolha nomes de secrets estáveis.
select vault.create_secret('https://<PROJECT_REF>.supabase.co', 'sga_project_url');
select vault.create_secret('<SUPABASE_SERVICE_ROLE_KEY>', 'sga_service_role_key');

-- Recurring sends for Meta approved templates through Zernio.
select cron.unschedule(jobid)
from cron.job
where jobname = 'sga-zernio-message-scheduler';

select cron.schedule(
  'sga-zernio-message-scheduler',
  '*/5 * * * *',
  $$
    select net.http_post(
      url := (select decrypted_secret from vault.decrypted_secrets where name = 'sga_project_url')
             || '/functions/v1/zernio-message-scheduler',
      headers := jsonb_build_object(
        'Content-Type', 'application/json',
        'apikey', (select decrypted_secret from vault.decrypted_secrets where name = 'sga_service_role_key')
      ),
      body := '{}'::jsonb
    ) as request_id;
  $$
);

-- Remove the antiga rotina com endpoint inválido e substitui o job gerenciado.
select cron.unschedule(jobid)
from cron.job
where jobname in ('processar-faturamento-diario', 'sga-faturamento-cobrancas');

-- A função existente é `faturamento-cobrancas`.
-- Rodar de hora em hora também recupera contratos que ficaram vencidos durante
-- indisponibilidade. Sem userId no corpo, a Edge Function processa todos os locadores.
select cron.schedule(
  'sga-faturamento-cobrancas',
  '*/5 * * * *',
  $$
    select net.http_post(
      url := (select decrypted_secret from vault.decrypted_secrets where name = 'sga_project_url')
             || '/functions/v1/faturamento-cobrancas',
      headers := jsonb_build_object(
        'Content-Type', 'application/json',
        'apikey', (select decrypted_secret from vault.decrypted_secrets where name = 'sga_service_role_key')
      ),
      body := '{}'::jsonb
    ) as request_id;
  $$
);
